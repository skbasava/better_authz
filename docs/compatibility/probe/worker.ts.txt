export default {
  async fetch(): Promise<Response> {
    const results: Record<string, unknown> = {};

    // B-04: multi-value Set-Cookie must round-trip through getSetCookie().
    const headers = new Headers();
    headers.append("Set-Cookie", "a=1; Path=/; HttpOnly");
    headers.append("Set-Cookie", "b=2; Path=/; HttpOnly");
    const cookies = headers.getSetCookie();
    results.getSetCookie = {
      available: typeof headers.getSetCookie === "function",
      count: cookies.length,
      values: cookies,
      pass: typeof headers.getSetCookie === "function" && cookies.length === 2,
    };

    // getSession returnHeaders/asResponse behavior (gate B-06) is probed in
    // run-functional.mjs against a real pinned instance; see findings doc.

    return Response.json(results, { headers: { "Cache-Control": "no-store" } });
  },
};
